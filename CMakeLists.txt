cmake_minimum_required(VERSION 3.24)
project(IsaacProf VERSION 0.11.8 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "IsaacProf POC requires Windows x86. Configure with -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
FetchContent_Declare(minhook URL https://codeload.github.com/TsudaKageyu/minhook/zip/c3fcafdc10146beb5919319d0683e44e3c30d537 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(lua_src URL https://codeload.github.com/TeamREPENTOGON/lua/zip/252c5578d935ba53b2dd2e25da6ed2d30aa03161 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(minhook lua_src)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/bootstrap.lua" BOOTSTRAP)
configure_file(src/bootstrap.hpp.in generated/bootstrap.hpp @ONLY)
add_library(isaacprof SHARED src/profiler.cpp src/screenshots.cpp src/native_waits.cpp)
target_include_directories(isaacprof PRIVATE "${lua_src_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(isaacprof PRIVATE minhook opengl32 gdi32 ole32 windowscodecs psapi bcrypt version)
add_executable(isaacprof-injector src/injector.cpp)
add_executable(isaacprof-analyze src/analyze.cpp)
foreach(target isaacprof isaacprof-injector isaacprof-analyze)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
endforeach()
option(ISAACPROF_BUILD_TESTS "Build controlled Lua injection host" ON)
if(ISAACPROF_BUILD_TESTS)
  file(GLOB LUA_SOURCES "${lua_src_SOURCE_DIR}/l*.c")
  list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|ltests)\\.c$")
  add_library(test-lua SHARED ${LUA_SOURCES})
  target_compile_definitions(test-lua PRIVATE LUA_BUILD_AS_DLL)
  # Current REPENTOGON ships Lua 5.4 under this legacy filename.
  set_target_properties(test-lua PROPERTIES OUTPUT_NAME "Lua5.3.3r")
  add_executable(isaacprof-test-host tests/host.cpp)
  target_include_directories(isaacprof-test-host PRIVATE "${lua_src_SOURCE_DIR}")
  target_link_libraries(isaacprof-test-host PRIVATE test-lua opengl32 gdi32)
  target_compile_definitions(isaacprof-test-host PRIVATE LUA_BUILD_AS_DLL WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(isaacprof-test-host PRIVATE /W4 /WX /utf-8 /EHsc)
endif()
