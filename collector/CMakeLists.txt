cmake_minimum_required(VERSION 3.24)
project(IsaacProfCollector VERSION 0.8.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Collector requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(isaacprof-collector ../src/collector.cpp)
target_compile_definitions(isaacprof-collector PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(isaacprof-collector PRIVATE /W4 /WX /utf-8 /EHsc)
